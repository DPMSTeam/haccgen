    <script>
        try {
            document.addEventListener('DOMContentLoaded', function() {
                const topicsWrapper = document.getElementById('topics-wrapper');
                const topicOrderInput = document.getElementById('topic-order');
                const addTopicBtn = document.getElementById('add-topic-btn');
                const form = document.getElementById('haccgen-form');

                // Event delegation for topic actions
                topicsWrapper.addEventListener('click', (e) => {
                    const item = e.target.closest('.topic-item');
                    if (!item) return;

                    const deleteBtn = e.target.closest('.delete-btn');
                    const editBtn = e.target.closest('.edit-btn');

                    if (deleteBtn) {
                        item.remove();
                        updateTopicOrder();
                    } else if (editBtn) {
                        const span = item.querySelector('span');
                        const currentText = span.textContent;
                        const topicLabel = currentText.split(':')[0];
                        const topicTitle = currentText.split(':').slice(1).join(':').trim();

                        const input = document.createElement('input');
                        input.type = 'text';
                        input.value = topicTitle;
                        input.className = 'form-control';
                        span.replaceWith(input);
                        input.focus();
                        
                   // Disable save button during editing
                document.getElementById('save-btn').disabled = true;

                input.addEventListener('blur', () => {
                const newTitle = input.value.trim();
                const newSpan = document.createElement('span');
                newSpan.textContent = `${topicLabel}: ${newTitle || topicTitle}`;
                input.replaceWith(newSpan);

                updateTopicOrder();

                 // Re-enable save button when editing is done
                 document.getElementById('save-btn').disabled = false;
                 });


                        input.addEventListener('blur', () => {
                            const newTitle = input.value.trim();
                            if (newTitle) {
                                const newSpan = document.createElement('span');
                                newSpan.textContent = `${topicLabel}: ${newTitle}`;
                                input.replaceWith(newSpan);
                                updateTopicOrder();
                            } else {
                                input.replaceWith(span);
                            }
                        });

                        input.addEventListener('keypress', (e) => {
                            if (e.key === 'Enter') input.blur();
                        });
                    }
                });

                // Drag-and-drop
                topicsWrapper.addEventListener('dragstart', (e) => {
                    const item = e.target.closest('.topic-item');
                    if (item && item.dataset.index !== undefined) {
                        e.dataTransfer.setData('text/plain', item.dataset.index);
                    }
                });

                topicsWrapper.addEventListener('dragover', (e) => e.preventDefault());

                topicsWrapper.addEventListener('drop', (e) => {
                    e.preventDefault();
                    const fromIndex = e.dataTransfer.getData('text/plain');
                    const toItem = e.target.closest('.topic-item');
                    if (!toItem || fromIndex === toItem.dataset.index) return;

                    const topics = document.querySelectorAll('.topic-item');
                    const [movedItem] = Array.from(topics).splice(fromIndex, 1);
                    topicsWrapper.insertBefore(movedItem, toItem);

                    Array.from(topics).forEach((item, index) => {
                        item.dataset.index = index;
                        const span = item.querySelector('span');
                        const title = span.textContent.split(':').slice(1).join(':').trim();
                        span.textContent = `Topic ${index + 1}: ${title}`;
                    });

                    updateTopicOrder();
                });

                // Keyboard navigation
                topicsWrapper.addEventListener('keydown', (e) => {
                    const item = e.target.closest('.topic-item');
                    if (!item) return;

                    if (e.key === 'ArrowUp' && item.previousElementSibling) {
                        topicsWrapper.insertBefore(item, item.previousElementSibling);
                        updateTopicOrder();
                    } else if (e.key === 'ArrowDown' && item.nextElementSibling) {
                        topicsWrapper.insertBefore(item.nextElementSibling, item);
                        updateTopicOrder();
                    }
                });

                // Add topic
                addTopicBtn.addEventListener('click', () => {
                    const newIndex = document.querySelectorAll('.topic-item').length;
                    const newTopic = `
                        <div class="topic-item" draggable="true" data-index="${newIndex}" data-content="" data-id="topic-${newIndex}" tabindex="0">
                            <span>Topic ${newIndex + 1}: New Topic</span>
                            <div>
                                <button type="button" class="move-btn"><i class="fas fa-arrows-alt"></i></button>
                                <button type="button" class="edit-btn"><i class="fas fa-pencil-alt"></i></button>
                                <button type="button" class="delete-btn"><i class="fas fa-trash"></i></button>
                            </div>
                        </div>
                    `;
                    topicsWrapper.insertAdjacentHTML('beforeend', newTopic);
                    updateTopicOrder();
                });

                // Update topic order
                function updateTopicOrder() {
                    const order = Array.from(topicsWrapper.querySelectorAll('.topic-item')).map((item, index) => ({
                        id: item.dataset.id || `topic-${index}`,
                        title: item.querySelector('span').textContent.replace(`Topic ${index + 1}: `, '').trim(),
                        content: item.dataset.content || ''
                    }));
                    try {
                        topicOrderInput.value = JSON.stringify(order);
                        console.log('Updated topic_order:', topicOrderInput.value);
                    } catch (error) {
                        console.error('Failed to stringify topic order:', error);
                        topicOrderInput.value = '[]';
                        alert('Error updating topics. Please try again.');
                    }
                }

                // Debug form submission
                form.addEventListener('submit', (e) => {
                    console.log('Form submitting with topic_order:', topicOrderInput.value);
                });

                updateTopicOrder();
            });
        } catch (error) {
            console.error('JavaScript error in haccgen_template:', error);
            alert('An error occurred while managing topics. Please try again.');
        }
    </script>
<script>
document.addEventListener('DOMContentLoaded', function () {
    const form = document.getElementById('haccgen-form');
    const saveBtn = document.getElementById('save-btn');
    const backBtn = form.querySelector('button[name="action"][value="back"]');
     const addTopicBtn = document.getElementById('add-topic-btn');


    if (!form || !saveBtn) return;

    // Spinner markup to reuse
    const spinnerHTML = `
        <span class="spinner-border spinner-border-sm me-2" role="status" aria-hidden="true"></span>
        Saving...
    `;

    form.addEventListener('submit', function (e) {
        const submitter = e.submitter || document.activeElement;

        // Only show spinner on Save button, not on Back
        if (submitter && submitter === saveBtn) {
            setTimeout(() => {
                saveBtn.disabled = true;
                saveBtn.innerHTML = spinnerHTML;

                // Optional: also disable the back button
                if (backBtn) backBtn.disabled = true;
                if (addTopicBtn) addTopicBtn.disabled = true;
            }, 0); // allow form to submit first before blocking UI
        }
    });
});
</script>
